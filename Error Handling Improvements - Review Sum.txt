  }

      }

      return nil  // ❌ Forces nil checks everywhere
  }

  After:
  // GetAppError extracts an AppError from the error chain.
  // If the error is not an AppError, it returns a default internal server error
  // with the original error wrapped for debugging purposes.
  // This function never returns nil.
  func GetAppError(err error) *AppError {
      if err == nil {
          return Internal()
      }

      var appErr *AppError
      if errors.As(err, &appErr) {
          return appErr
      }

      // ✅ Return a default internal error with the original error wrapped
      return InternalWithErr(err)
  }

  5. Enhanced Postgres Error Mapping with Constraint Names

  Problem: Generic "constraint violation" errors provided no helpful information.

  /internal/activity/application/error_mapper.go:78

  /internal/account-profile/application/account_profile_service.go:411

  Before:
  func MapPostgresError(err *pgconn.PgError) *domainerror.AppError {
      if pgerrcode.IsIntegrityConstraintViolation(err.Code) {
          return domainerror.ValidationError(
              "unique_violation",
              "Unique constraint violation",  // ❌ Generic message
              err,
          )
      }

      return domainerror.Internal()
  }

  After:
  func MapPostgresError(err *pgconn.PgError) *domainerror.AppError {
      // Unique constraint violations
      if err.Code == pgerrcode.UniqueViolation {
          return domainerror.Conflict(
              "unique_violation",
              formatConstraintMessage(err, "already exists"),  // ✅ "email already exists"
              err,
          )
      }

      // Foreign key violations
      if err.Code == pgerrcode.ForeignKeyViolation {
          return domainerror.ValidationError(
              "foreign_key_violation",
              formatConstraintMessage(err, "references invalid or missing record"),
              err,
          )
      }

      // Not-null constraint violations
      if err.Code == pgerrcode.NotNullViolation {
          return domainerror.ValidationError(
              "not_null_violation",
              formatConstraintMessage(err, "is required"),  // ✅ "password is required"
              err,
          )
      }

      // Check constraint violations
      if err.Code == pgerrcode.CheckViolation {
          return domainerror.ValidationError(
              "check_violation",
              formatConstraintMessage(err, "violates check constraint"),
              err,
          )
      }

      // Other integrity constraint violations
      if pgerrcode.IsIntegrityConstraintViolation(err.Code) {
          return domainerror.ValidationError(
              "constraint_violation",
              formatConstraintMessage(err, "violates database constraint"),
              err,
          )
      }

      return domainerror.InternalWithErr(err)
  }

  // ✅ NEW HELPER FUNCTION
  func formatConstraintMessage(err *pgconn.PgError, defaultSuffix string) string {
      // Try column name first
      if err.ColumnName != "" {
          return err.ColumnName + " " + defaultSuffix
      }

      // Try constraint name
      if err.ConstraintName != "" {
          return err.ConstraintName + " " + defaultSuffix
      }

      // Try detail message
      if err.Detail != "" {
          return err.Detail
      }

      return "Database constraint violation"
  }

  - Foreign key: "account_id references invalid or missing record" instead of "Unique constraint violation"
  - Not-null: "password is required" instead of "Unique constraint violation"
  - Check: "age violates check constraint" instead of "Unique constraint violation"

  6. Validation Error Details in Responses

  Problem: JSON decode errors returned generic "Invalid request body" with no field details.

  /internal/shared/api/request.go (Completely Rewritten)

  After:
  package api

  import (
      "encoding/json"
      "errors"
      "fmt"
      "io"
      "net/http"
      "strings"
  )

  // ✅ NEW: ValidationError represents field-level validation errors
  type ValidationError struct {
      Message string
      Fields  map[string]string // field -> error message
  }

  func (e *ValidationError) Error() string {
      return e.Message
  }

  // ✅ ENHANCED: DecodeJSONBody decodes the request body and returns detailed validation errors
  func DecodeJSONBody(r *http.Request, body any) error {
      decoder := json.NewDecoder(r.Body)
      decoder.DisallowUnknownFields() // ✅ Reject unknown fields

      err := decoder.Decode(body)
      if err == nil {
          return nil
      }

      // ✅ Parse JSON decode errors into field-level details
      return parseJSONDecodeError(err)
  }

  // ✅ NEW: parseJSONDecodeError converts JSON decode errors into structured ValidationError
  func parseJSONDecodeError(err error) error {
      var syntaxError *json.SyntaxError
      var unmarshalTypeError *json.UnmarshalTypeError

      switch {
      // Syntax errors (malformed JSON)
      case errors.As(err, &syntaxError):
          return &ValidationError{
              Message: "Invalid JSON syntax",
              Fields: map[string]string{
                  "_json": fmt.Sprintf("Syntax error at position %d", syntaxError.Offset),
              },
          }

      // Type mismatch errors
      case errors.As(err, &unmarshalTypeError):
          fieldName := unmarshalTypeError.Field
          if fieldName == "" {
              fieldName = "unknown"
          }
          return &ValidationError{
              Message: "Invalid field type",
              Fields: map[string]string{
                  fieldName: fmt.Sprintf("Expected %s but got %s",
                      unmarshalTypeError.Type.String(),
                      unmarshalTypeError.Value),
              },
          }

      // Unknown fields error
      case strings.Contains(err.Error(), "unknown field"):
          // Extract field name from error message
          msg := err.Error()
          if start := strings.Index(msg, "\""); start != -1 {
              if end := strings.Index(msg[start+1:], "\""); end != -1 {
                  fieldName := msg[start+1 : start+1+end]
                  return &ValidationError{
                      Message: "Unknown field in request",
                      Fields: map[string]string{
                          fieldName: "This field is not recognized",
                      },
                  }
              }
          }
          // ... fallback

      // EOF error (empty body)
      case errors.Is(err, io.EOF):
          return &ValidationError{
              Message: "Request body is empty",
              Fields: map[string]string{
                  "_body": "Request body cannot be empty",
              },
          }

      // Unexpected EOF (incomplete JSON)
      case errors.Is(err, io.ErrUnexpectedEOF):
          return &ValidationError{
              Message: "Incomplete JSON",
              Fields: map[string]string{
                  "_json": "JSON is incomplete or malformed",
              },
          }

      default:
          return &ValidationError{
              Message: "Invalid request body",
              Fields: map[string]string{
                  "_request": err.Error(),
              },
          }
      }
  }

  // ✅ NEW: WriteValidationErrorResponse writes a validation error response with field details
  func WriteValidationErrorResponse(w http.ResponseWriter, err error) error {
      var validationErr *ValidationError
      if errors.As(err, &validationErr) {
          details := make(map[string]any)
          for field, msg := range validationErr.Fields {
              details[field] = msg
          }
          return WriteBadRequestWithDetails(
              w,
              "validation_error",
              validationErr.Message,
              details,
          )
      }

      // Fallback to generic bad request
      return WriteBadRequestResponse(w, "bad_request", "Invalid request body")
  }
